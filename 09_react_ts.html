Best Practices

Плохо vs хорошо

плохо

const Button = (props: any) => {
    return <button onClick={props.onClick}>...;</button>
};

хорошо

const Button: React.FC<ButtonProps> = ({
    onClick, children
}) => {
    return <button onClick={onClick}>{children}</button>;
};
  • interface для сложных объектов, избегать any
  • Union-типы, кастомные хуки с типами, события — типизированы
Типизация компонентов

Card: interface CardProps

2/Card.tsx

import React from 'react'
import './Card.css'

export interface CardProps {
  title?: string;
  children: React.ReactNode;
  className?: string;
}

export const Card: React.FC<CardProps> =
({ title, children, className = '' }) => {
  return (
    <div className={`card ${className}`}>
      {title && <h3 className="card__title">{title}</h3>}
      <div className="card__content">{children}</div>
    </div>
  )
}

children типизируются как React.ReactNode.

Типизация компонентов

Button: variants + CSS Modules

3/Button.tsx

import React from 'react'
import styles from './Button.module.css'

export interface ButtonProps {
  variant?: 'primary' | 'secondary';  // union types
  size?: 'small' | 'medium' | 'large';
  type?: 'submit' | 'button';
  disabled?: boolean;
  onClick?: () => void;               // функция в props
  children: React.ReactNode;
}

export const Button: React.FC<ButtonProps> = ({
  variant = 'primary', size = 'medium',
  disabled = false, type = 'button',
  onClick, children,
}) => {
  const className = [
    styles.btn,
    styles[`btn--${variant}`],
    styles[`btn--${size}`],
    disabled ? styles.disabled : '',
  ].join(' ')

  return (
    <button className={className} disabled={disabled}
            onClick={onClick} type={type}>
      {children}
    </button>
  )
}

CSS Modules: import styles from './Button.module.css'.

App.tsx

6/App.tsx

function App(): JSX.Element {
  const [count, setCount] = useState(0)

  const handleButtonClick = () => {
    setCount(prev => prev + 1)
  }

  return (
    <main className="app-main">
      <Card title='Это заголовок' className='main-card'>
        Счетчик: {count}
        <Button variant='secondary' onClick={handleButtonClick}>
          Кнопка
        </Button>
      </Card>
    </main>
  )
}
useState

Типизированное состояние

// строка/число — тип выводится сам
const [count, setCount] = useState<number>(0);

// объект | null — задача с планами
const [user, setUser] = useState<User | null>(null);

// массив
const [users, setUsers] = useState<User[]>([]);

// union для статусов загрузки
const [status, setStatus] =
    useState<'loading' | 'success' | 'error'>('loading');

Дженерик useState<···> — точно описываем, что храним.
