Синхронный vs асинхронный

Почему async нужен

1.js

// Синхронно: построчно, предсказуемо
console.log('Начало');
console.log('Середина');
console.log('Конец');

// Асинхронно: setTimeout не блокирует код
console.log('Начало');
setTimeout(() => console.log('Середина'), 1000);
console.log('Конец');   // выполнится СРАЗУ, не через секунду
  • UI остаётся отзывчивым, пока «в фоне» идёт работа
  • Та же модель: события, таймеры, сетевые запросы

Модель выполнения

  1. Call Stack — стек вызовов, один поток: одна операция за раз
  2. Web APIs — браузер выполняет таймеры/запросы «в фоне»
  3. Callback Queue — очередь готовых колбэков
  4. Event Loop — пустой стек? → переносит колбэк в стек
Event Loop в действии

Почему setTimeout(…, 0) не мгновенный

2.js

console.log('Начало');

setTimeout(() => console.log('Таймер'), 0);

console.log('Конец');

Вывод: «Начало», «Конец», «Таймер».

  • Колбэк попадает в Callback Queue и ждёт пустого стека
  • Отложен на минимум 0 мс — но не раньше конца синхронного кода
Таймеры

setTimeout · setInterval

3.js

// Однократно через 2 секунды
const timeoutId = setTimeout(() => {
    console.log('Выполнится через 2 секунды');
}, 2000);

// Каждую секунду
const intervalId = setInterval(() => {
    console.log('Выполняется каждую секунду');
}, 1000);

// Остановка по id
// clearTimeout(timeoutId);
// clearInterval(intervalId);

setTimeout/setInterval возвращают id — по нему таймер можно отменить.

Callbacks

Функция, которую вызовут позже

4.js

// Синхронный callback
function doHomework(subject, callback) {
    console.log(`Начинаю делать лабу по ${subject}`);
    callback();
}
function finishedHomework() {
    console.log('Закончил делать домашку');
}
doHomework('js', finishedHomework);

// Асинхронный callback — выполнится по клику
document.getElementById('button').addEventListener('click', function() {
    console.log('Кнопка нажата!');
});

Промисы

Обещание результата

5.js

const promise = new Promise((resolve, reject) => {
    // Асинхронная операция
    setTimeout(() => {
        const success = Math.random() > 0.5;
        if (success) {
            resolve('Данные успешно получены');
        } else {
            reject('Ошибка загрузки данных');
        }
    }, 1000);
});
  • Три состояния: pending → fulfilled / rejected
  • resolve(value) — успех, reject(error) — ошибка
Промисы

Цепочки .then/.catch/.finally

getUser(userId)                       // промис
    .then(user => getPosts(user.id))
    .then(posts => getComments(posts[0].id))
    .then(comments => getLikes(comments[0].id))
    .then(likes => console.log(likes))
    .catch(error => console.error(error))  // одна точка ошибок
    .finally(() => console.log('Готово')); // всегда
  • Каждый .then возвращает промис → цепочка продолжается
  • catch ловит ошибку из любого звена
Промисы

Promise.all · race · allSettled

МетодПоведение
Promise.allждёт ВСЕ — результат массивом; упал один = ошибка
Promise.raceпервый завершившийся (успех или ошибка)
Promise.allSettledвсе с исходом каждого, даже с ошибками

Fetch API

6.js

// GET
fetch('https://api.hh.ru/cors')
    .then(response => {
        if (!response.ok) throw new Error('Ошибка сети');
        return response.json();
    })
    .then(data => console.log(data))
    .catch(error => console.error('Ошибка:', error));

// POST
fetch('https://api.hh.ru/cors', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ name: 'John', age: 30 })
})
.then(response => response.json())
.then(data => console.log(data));

Async / Await

7.js

// Тот же fetch, но читается как синхронный код
async function loadData() {
    try {
        const response = await fetch('https://api.hh.ru/cors');
        if (!response.ok) throw new Error('Ошибка HTTP: ' + response.status);
        const data = await response.json();
        console.log(data);
    } catch (error) {
        console.error('Не удалось загрузить:', error);
    } finally {
        console.log('Запрос завершен');
    }
}

loadData();
