Функции

Замыкания

5.js

function createCounter() {
    let count = 0; // эта переменная — «приватная»
    return function() {
        count++;       // функция «запоминает» окружение создания
        return count;
    };
}

const counter = createCounter();
counter(); // 1
counter(); // 2 — прямой доступ к count извне недоступен

Замыкание = функция + её лексическое окружение.

Функции

Каррирование

6.js

function curriedMultiply(a) {
    return function(b) {
        return function(c) {
            return a * b * c;
        };
    };
}

curriedMultiply(2)(3)(4); // 24

// специализированная функция через замыкание
const multiplyByTwo = curriedMultiply(2);
multiplyByTwo(3)(4); // 24

Геттеры, сеттеры, статика

10.js

class User {
    constructor(firstName, lastName) {
        this.firstName = firstName;
        this.lastName = lastName;
    }
    get fullName() {              // вызывается как свойство
        return `${this.firstName} ${this.lastName}`;
    }
    set fullName(newValue) {      // при присваивании
        [this.firstName, this.lastName] = newValue.split(' ');
    }
    static createGuest() {        // вызов через имя класса
        return new User("Гость", "Анонимный");
    }
}

user.fullName        // "Иван Петров"  (геттер, без скобок!)
user.fullName = "Алексей Сидоров"; // сеттер
User.createGuest();                // статический метод

Наследование

11.js

class Animal {
    constructor(name) {
        this.name = name;
        this.speed = 0;
    }
    run(speed) {
        this.speed = speed;
        console.log(`${this.name} бежит со скоростью ${this.speed}.`);
    }
}

class Cat extends Animal {   // extends — наследует от Animal
    constructor(name, owner) {
        super(name);         // super() — конструктор родителя
        this.owner = owner;
    }
    meow() { console.log(`${this.name} говорит Мяу!`); }
    run(speed) {             // переопределение метода
        super.run(speed * 2);
        console.log("И делает это изящно!");
    }
}

const myCat = new Cat("Барсик", "Мария");
myCat.meow();   // Барсик говорит Мяу!
myCat.run(5);   // Барсик бежит со скоростью 10. И делает это изящно!

Массивы

for и for…of

14.js

const fruits = ['apple', 'banana', 'orange'];

for (let i = 0; i < fruits.length; i++) {
    console.log(`${i}: ${fruits[i]}`);

Методы массивов

reduce — аккумуляция

18.js

// сумма
const sum = numbers.reduce((acc, n) => acc + n, 0); // 15

// подсчёт количества — аккумулятор-объект
const count = fruits.reduce((acc, fruit) => {
    acc[fruit] = (acc[fruit] || 0) + 1;
    return acc;
}, {});
// {apple: 3, banana: 2, orange: 1}

// максимум
const max = numbers.reduce((acc, n) => Math.max(acc, n), -Infinity); // 5
