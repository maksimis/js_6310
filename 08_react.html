Что такое React

  • JS-библиотека для построения UI (Meta, 2013)
  • Компонентный подход — интерфейс из переиспользуемых частей
  • Декларативность: описываем, как должен выглядеть UI, React сам обновляет DOM
  • Однонаправленный поток данных: вниз — через props, наверх — через колбэки (onChange, onSubmit)

Real vs Virtual DOM

  • Браузерный DOM — императивно: сами ищем элементы и меняем их (innerHTML, appendChild)
  • Virtual DOM — лёгкие JS-объекты: React сравнивает старое и новое дерево и меняет в DOM только разницу. Он не «быстрее DOM», он избавляет от ручных обновлений
  • React сам решает, что обновить: Update → Reconciliation → New Virtual DOM → Browser DOM

const users = [
    { id: 1, name: "Анна Петрова", age: 22, email: "anna@example.com", isOnline: true },
    { id: 2, name: "Борис Иванов", age: 25, email: "boris@example.com" },
];

function App() {
    return (
        <div>
            {users.map(user => (
                <UserCard
                    key={user.id}          // стабильный уникальный id
                    name={user.name}
                    age={user.age}
                    email={user.email}
                    isOnline={user.isOnline}
                />
            ))}
        </div>
    );
}
  • key помогает React понять, какой элемент добавили, удалили или переставили
  • Ключ — id из данных. Индекс массива (key={i}) ломается при удалении и сортировке: состояние «переезжает» к соседнему элементу
  • Без key React выводит предупреждение в консоль

useState

08_react/5_useState/src/components/Counter.jsx

import { useState } from 'react';

function Counter() {
    const [count, setCount] = useState(0);
    const [name, setName] = useState('');

    // новое значение считаем из предыдущего: так безопасно
    // при нескольких обновлениях подряд
    const increment = () => setCount(prev => prev + 1);
    const decrement = () => setCount(prev => prev - 1);
    const reset = () => setCount(0);

    return (
        <div className="counter">
            <h2>Счетчик: {count}</h2>
            <input type="text" placeholder="Введите имя"
                   value={name}
                   onChange={(e) => setName(e.target.value)} />
            <p>Привет, {name || 'незнакомец'}!</p>
            <button onClick={decrement}>-</button>
            <button onClick={reset}>Сброс</button>
            <button onClick={increment}>+</button>
        </div>
    );
}

Изменяем состояние ТОЛЬКО через set-функцию — это запускает ре-рендер. setCount(count + 1) дважды подряд прибавит 1, а не 2: оба вызова видят старый count.

08_react/6_forms/src/components/LoginForm.jsx

function LoginForm() {
    const [formData, setFormData] = useState({
        email: '', password: '', rememberMe: false
    });

    const handleChange = (e) => {
        const { name, value, type, checked } = e.target;
        setFormData(prev => ({
            ...prev,
            [name]: type === 'checkbox' ? checked : value
        }));
    };

    const handleSubmit = (e) => {
        e.preventDefault();
        console.log('Данные формы:', formData);
    };

    return (
        <form onSubmit={handleSubmit}>
            <label htmlFor="email">Email:</label>
            <input id="email" type="email" name="email"
                   value={formData.email} onChange={handleChange} required />
            <label htmlFor="password">Пароль:</label>
            <input id="password" type="password" name="password"
                   value={formData.password} onChange={handleChange} required />
            <label>
                <input type="checkbox" name="rememberMe"
                       checked={formData.rememberMe} onChange={handleChange} />
                Запомнить меня
            </label>
            <button type="submit">Войти</button>
        </form>
    );
}
  • value + onChange, а не defaultValue — React «владеет» состоянием
  • Один handleChange на все поля через name
  • <label htmlFor> + id — подпись связана с полем: клик по ней ставит фокус, скринридер её читает

Загрузка данных

08_react/7_useEffect/src/components/UserProfile.jsx (сокращено)

function UserProfile({ userId }) {
    const [user, setUser] = useState(null);
    const [loading, setLoading] = useState(false);
    const [error, setError] = useState(null);

    useEffect(() => {
        if (!userId) return;
        let ignore = false;               // ответ для устаревшего userId не записываем

        const fetchUser = async () => {
            setLoading(true);
            setError(null);
            try {
                const response = await fetch(`https://jsonplaceholder.typicode.com/users/${userId}`);
                if (!response.ok) throw new Error(`HTTP ${response.status}`);  // fetch не бросает на 404
                const userData = await response.json();
                if (!ignore) setUser(userData);
            } catch (err) {
                if (!ignore) setError(err.message);
            } finally {
                if (!ignore) setLoading(false);
            }
        };
        fetchUser();

        return () => { ignore = true; };  // очистка: при смене userId или размонтировании
    }, [userId]);                         // зависимость: эффект при изменении userId

    if (!userId) return <div>Пользователь не выбран</div>;
    if (loading) return <div>Загрузка...</div>;
    if (error) return <div>Ошибка: {error}</div>;
    return user && <h1>{user.name}</h1>;
}

Зависимости [···]: пусто — только при монтировании, с данными — при их изменении. Подробнее про режимы и StrictMode — в теме 12.

Подъём состояния и children

пример

// Состояние нужно двум компонентам — храним его в общем родителе
function TemperatureApp() {
    const [celsius, setCelsius] = useState(20);

    return (
        <Panel title="Температура">
            <CelsiusInput value={celsius} onChange={setCelsius} />   {/* данные вниз, событие вверх */}
            <p>По Фаренгейту: {celsius * 9 / 5 + 32}</p>
        </Panel>
    );
}

function CelsiusInput({ value, onChange }) {
    return <input type="number" value={value} onChange={e => onChange(Number(e.target.value))} />;
}

// Композиция: Panel не знает, что внутри, — отображает children
function Panel({ title, children }) {
    return <section className="panel"><h2>{title}</h2>{children}</section>;
}
  • Состояние живёт в ближайшем общем родителе — «подъём состояния»
  • children — универсальные обёртки (Card, Layout, Modal) без копирования кода
